TEIL 3: JavaScript
Füge diesen Block ganz unten in index.html vor </body> ein,
am besten unter das Skript, das du schon für den Seitenanfang hast
(nur einmal, egal wie viele Player du hast)
=====================================================================

<script>
document.querySelectorAll('.sound').forEach(function (box) {
  var audio = box.querySelector('audio');
  var btn = box.querySelector('.play');
  var time = box.querySelector('.time');
  function fmt(s) { s = Math.floor(s || 0); return Math.floor(s / 60) + ':' + String(s % 60).padStart(2, '0'); }
  function show() { time.textContent = fmt(audio.currentTime) + (isFinite(audio.duration) ? ' / ' + fmt(audio.duration) : ''); }
  btn.addEventListener('click', function () {
    if (audio.paused) {
      document.querySelectorAll('.sound audio').forEach(function (a) { if (a !== audio) a.pause(); });
      audio.play().catch(function () {});
    } else {
      audio.pause();
    }
  });
  audio.addEventListener('play', function () { box.classList.add('is-playing'); btn.setAttribute('aria-label', 'Pause'); });
  audio.addEventListener('pause', function () { box.classList.remove('is-playing'); btn.setAttribute('aria-label', 'Abspielen'); });
  audio.addEventListener('ended', function () { audio.currentTime = 0; show(); });
  audio.addEventListener('timeupdate', show);
  audio.addEventListener('loadedmetadata', show);
});
</script>
